276impl From<Event> for EventKind {
277    #[inline]
278    fn from(mut event: Event) -> Self {
279        unsafe {
280            match event.get_statechange() {
281                StateChange::Combat => Self::Combat(event.extract()),
282                StateChange::EnterCombat => Self::EnterCombat(event.extract()),
283                StateChange::ExitCombat => Self::ExitCombat(event.extract()),
284                StateChange::ChangeUp => Self::ChangeUp(event.extract()),
285                StateChange::ChangeDead => Self::ChangeDead(event.extract()),
286                StateChange::ChangeDown => Self::ChangeDown(event.extract()),
287                StateChange::Spawn => Self::Spawn(event.extract()),
288                StateChange::Despawn => Self::Despawn(event.extract()),
289                StateChange::HealthUpdate => Self::HealthUpdate(event.extract()),
290                StateChange::SquadCombatStart => Self::SquadCombatStart(event.extract()),
291                StateChange::SquadCombatEnd => Self::SquadCombatEnd(event.extract()),
292                StateChange::WeaponSwap => Self::WeaponSwap(event.extract()),
293                StateChange::MaxHealthUpdate => Self::MaxHealthUpdate(event.extract()),
294                StateChange::PointOfView => Self::PointOfView(event.extract()),
295                StateChange::Language => Self::Language {
296                    time: event.time,
297                    language: u8::try_from(event.src_agent)
298                        .ok()
299                        .and_then(|value| Language::try_from(value).ok())
300                        .ok_or(event.src_agent),
301                },
302                StateChange::GWBuild => Self::GWBuild {
303                    time: event.time,
304                    build: event.src_agent,
305                },
306                StateChange::ShardId => Self::ShardId {
307                    time: event.time,
308                    shard: event.src_agent,
309                },
310                StateChange::Reward => Self::Reward(event.extract()),
311                StateChange::BuffInitial => Self::BuffInitial(event.extract()),
312                StateChange::Position => Self::Position(event.extract()),
313                StateChange::Velocity => Self::Velocity(event.extract()),
314                StateChange::Facing => Self::Facing(event.extract()),
315                StateChange::TeamChange => Self::TeamChange(event.extract()),
316                StateChange::AttackTarget => Self::AttackTarget(event.extract()),
317                StateChange::Targetable => Self::Targetable(event.extract()),
318                StateChange::MapId => Self::MapId {
319                    time: event.time,
320                    map: event.src_agent,
321                },
322                StateChange::StackActive => Self::StackActive(event.extract()),
323                StateChange::StackReset => Self::StackReset(event.extract()),
324                StateChange::Guild => Self::Guild(event.extract()),
325                StateChange::BuffInfo => Self::BuffInfo(event.extract()),
326                StateChange::BuffFormula => Self::BuffFormula(event.extract()),
327                StateChange::SkillInfo => Self::SkillInfo(event.extract()),
328                StateChange::SkillTiming => Self::SkillTiming(event.extract()),
329                StateChange::BreakbarState => Self::BreakbarState(event.extract()),
330                StateChange::BreakbarPercent => Self::BreakbarPercent(event.extract()),
331                StateChange::Integrity => Self::Integrity(event.extract()),
332                StateChange::Marker => Self::AgentMarker(event.extract()),
333                StateChange::BarrierUpdate => Self::BarrierUpdate(event.extract()),
334                StateChange::StatReset => Self::StatReset {
335                    time: event.time,
336                    target: event.src_agent,
337                },
338                StateChange::Extension => Self::Extension {
339                    sig: event.get_pad_id(),
340                    event,
341                },
342                StateChange::ApiDelayed => {
343                    event.is_statechange = StateChange::Combat.into();
344                    Self::ApiDelayed {
345                        event: event.into_kind().into(),
346                    }
347                }
348                StateChange::InstanceStart => Self::InstanceStart {
349                    time: event.time,
350                    start: event.src_agent,
351                },
352                StateChange::RateHealth => Self::RateHealth {
353                    time: event.time,
354                    rate: event.src_agent,
355                },
356                StateChange::Last90BeforeDown => Self::Last90BeforeDown(event.extract()),
357                StateChange::Effect45 => Self::Effect45(event.extract()),
358                StateChange::IdToGUID => Self::IdToGUID(event.extract()),
359                StateChange::LogNPCUpdate => Self::LogNPCUpdate(event.extract()),
360                StateChange::ExtensionCombat => Self::ExtensionCombat {
361                    sig: event.get_pad_id(),
362                    event,
363                },
364                StateChange::FractalScale => Self::FractalScale {
365                    time: event.time,
366                    scale: event.src_agent,
367                },
368                StateChange::Effect51 => Self::Effect51(event.extract()),
369                StateChange::Ruleset => Self::Ruleset(Ruleset::from_bits_retain(event.src_agent)),
370                StateChange::SquadMarker => Self::SquadMarker(event.extract()),
371                StateChange::ArcBuild => Self::ArcBuild(event.extract()),
372                StateChange::Glider => Self::Glider(event.extract()),
373                StateChange::Stunbreak => Self::Stunbreak(event.extract()),
374                StateChange::MissileCreate => Self::MissileCreate(event.extract()),
375                StateChange::MissileLaunch => Self::MissileLaunch(event.extract()),
376                StateChange::MissileRemove => Self::MissileRemove(event.extract()),
377                StateChange::EffectGroundCreate => Self::EffectGroundCreate(event.extract()),
378                StateChange::EffectGroundRemove => Self::EffectGroundRemove(event.extract()),
379                StateChange::EffectAgentCreate => Self::EffectAgentCreate(event.extract()),
380                StateChange::EffectAgentRemove => Self::EffectAgentRemove(event.extract()),
381                StateChange::IIDChange => Self::IIDChange {
382                    time: event.time,
383                    id: event.dst_agent,
384                    previous: event.src_agent,
385                },
386                StateChange::MapChange => Self::MapChange {
387                    time: event.time,
388                    map: event.src_agent,
389                    previous: event.dst_agent,
390                },
391                StateChange::AnimationStart => Self::AnimationStart(event.extract()),
392                StateChange::AnimationStop => Self::AnimationStop(event.extract()),
393                StateChange::BuffApply => Self::BuffApply(event.extract()),
394                StateChange::BuffChange => Self::BuffChange(event.extract()),
395                StateChange::BuffRemoveSingle => Self::BuffRemoveSingle(event.extract()),
396                StateChange::BuffRemoveAll => Self::BuffRemoveAll(event.extract()),
397                StateChange::Transformation => Self::Transformation(event.extract()),
398                StateChange::WvwTeams => Self::WvwTeams(event.extract()),
399                StateChange::WvwObjectiveStatus => Self::WvwObjectiveStatus(event.extract()),
400                StateChange::StealthChange => Self::StealthChange(event.extract()),
401                StateChange::GadgetAnimation => Self::GadgetAnimation(event.extract()),
402                StateChange::GadgetName => Self::GadgetName(event.extract()),
403                StateChange::IdleEvent | StateChange::ReplInfo | StateChange::EarlyExit => {
404                    unreachable!("illegal internal statechange")
405                }
406                StateChange::Unknown(_) => Self::Unknown(event),
407            }
408        }
409    }
