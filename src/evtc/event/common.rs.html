6/// Information common to combat events.
7#[derive(Debug, Clone)]
8#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
9pub struct CommonEvent {
10    /// Time of registering the event.
11    pub time: u64,
12
13    /// Agent that caused the event.
14    pub source: AgentId,
15
16    /// Agent the event happened to.
17    pub target: AgentId,
18
19    /// Skill id of the relevant skill (can be zero).
20    pub skill_id: u32,
21
22    /// Current affinity of source and target..
23    ///
24    /// *Arc calls this "iff" for if friend/foe.*
25    pub affinity: Affinity,
26
27    /// Whether source is above 90% Health.
28    pub is_ninety: bool,
29
30    /// Whether target is below 50% Health.
31    pub is_fifty: bool,
32
33    /// Whether source is moving at time of event.
34    pub is_source_moving: bool,
35
36    /// Whether target is moving at time of event.
37    pub is_target_moving: bool,
38
39    /// Whether source is flanking target at time of event.
40    ///
41    /// The value lies in a range of `1` to `135` degrees where `135` is rear.
42    pub is_flanking: u8,
43}
44
45impl From<&Event> for CommonEvent {
46    #[inline]
47    fn from(event: &Event) -> Self {
48        Self {
49            time: event.time,
50            source: AgentId::from_src(event),
51            target: AgentId::from_dst(event),
52            skill_id: event.skill_id,
53            affinity: event.get_affinity(),
54            is_ninety: event.is_ninety != 0,
55            is_fifty: event.is_fifty != 0,
56            is_source_moving: (event.is_moving & 1) != 0,
57            is_target_moving: (event.is_moving & 2) != 0,
58            is_flanking: event.is_flanking,
59        }
60    }
61}
