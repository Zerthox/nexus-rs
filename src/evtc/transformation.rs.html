1use crate::{AgentId, Event, StateChange, TryExtract, extract::Extract};
2
3#[cfg(feature = "serde")]
4use serde::{Deserialize, Serialize};

6/// Transformation change.
7#[derive(Debug, Clone)]
8#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
9pub struct TransformationEvent {
10    /// Time of registering the event.
11    pub time: u64,
12
13    /// Agent that caused the event.
14    pub source: AgentId,
15
16    /// Agent the event happened to.
17    pub target: AgentId,
18
19    /// Id of the transformation.
20    ///
21    /// Use to map to a GUID using [`StateChange::IdToGUID`] events.
22    pub transformation_id: u32,
23}
24
25impl Extract for TransformationEvent {
26    #[inline]
27    unsafe fn extract(event: &Event) -> Self {
28        Self {
29            time: event.time,
30            source: AgentId::from_src(event),
31            target: AgentId::from_dst(event),
32            transformation_id: event.skill_id,
33        }
34    }
35}
36
37impl TryExtract for TransformationEvent {
38    #[inline]
39    fn can_extract(event: &Event) -> bool {
40        event.get_statechange() == StateChange::Transformation
41    }
42}
