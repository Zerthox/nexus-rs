1//! Bindings for the ArcDPS EVTC API.
2//!
3//! Includes both types for Arc's realtime API used by plugins as well as Arc's log API consumed by parsers.
4//!
5//! ```no_run
6//! use evtc::event::{Event, CombatEvent};
7//!
8//! fn total_damage_dealt(source: u64, target: u64, events: &[Event]) -> i32 {
9//!     events
10//!         .iter()
11//!         .filter_map(|event| event.try_extract::<CombatEvent>())
12//!         .filter(|event| {
13//!             event.result.is_strike_damage()
14//!                 && event.source.id == source
15//!                 && event.target.id == target
16//!         })
17//!         .map(|event| event.non_shield_strike_damage())
18//!         .sum()
19//! }
20//! ```
